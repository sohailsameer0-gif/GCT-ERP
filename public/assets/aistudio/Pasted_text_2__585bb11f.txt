      var dupCnic=state.students.find(function(s){return String(s.cnic||"").replace(/\D/g,"")===cnicClean&&s.id!==cur;});
      if(dupCnic)e.cnic="CNIC already registered";
    }
  }
  if(d.dob){
    var dobDate=new Date(d.dob+"T00:00:00");
    if(isNaN(dobDate))e.dob="Invalid date";
    else{
      var today=new Date();
      if(dobDate>today)e.dob="DOB cannot be future";
      else{
        var age=today.getFullYear()-dobDate.getFullYear();
        var m=today.getMonth()-dobDate.getMonth();
        if(m<0||(m===0&&today.getDate()<dobDate.getDate()))age--;
        if(age<10)e.dob="Student must be at least 10 years old";
        if(age>80)e.dob="Invalid DOB";
      }
    }
  }
  if(d.email){
    var emailLower=String(d.email).trim().toLowerCase();
    if(!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(emailLower))e.email="Invalid email format";
    else{
      var dupEmail=state.students.find(function(s){return String(s.email||"").toLowerCase()===emailLower&&s.id!==cur;});
      if(dupEmail)e.email="Email already registered";
      var accEmail=(state.accounts||[]).find(function(a){return String(a.email||"").toLowerCase()===emailLower&&a.role!=="student";});
      if(accEmail)e.email="Email already used by another account";
    }
  }
  if(!d.programId)e.programId="Program required";
  if(!d.departmentId)e.departmentId="Department required";
  if(!d.currentSemesterId)e.currentSemesterId="Semester required";
  if(!d.currentSectionId)e.currentSectionId="Section required";
  if(d.programId&&!getProgram(d.programId))e.programId="Program not found";
  if(state.students.find(function(s){return s.admissionNumber===d.admissionNumber&&s.id!==cur;}))e.admissionNumber="Admission # exists";
  if(state.students.find(function(s){return s.registrationNumber===d.registrationNumber&&s.id!==cur;}))e.registrationNumber="Reg # exists";
  if(state.students.find(function(s){return s.classNumber===d.classNumber&&s.id!==cur;}))e.classNumber="Class # exists";
  return e;
}

function submitStudentLeave(session,data){
  if(session.role!=="student")throw new Error("Only students");
  if(!session.linkedId)throw new Error("Not linked");
  data.reason=String(data.reason||"").trim();
  if(!data.reason)throw new Error("Leave reason is required");
  var start=new Date(data.startDate+"T00:00:00"),end=new Date(data.endDate+"T00:00:00");
  if(start>end)throw new Error("Start must be before end");
  var days=Math.round((end-start)/86400000)+1;
  var stu=state.students.find(function(s){return s.id===session.linkedId;});
  if(!stu)throw new Error("Student record not found");
  var requestedDays={};var cur=new Date(start);
  while(cur<=end){requestedDays[dayOfWeek(cur)]=true;cur.setDate(cur.getDate()+1);}
  var relevantSlots=state.timetable.filter(function(t){return t.semesterId===stu.currentSemesterId&&t.sectionId===stu.currentSectionId&&requestedDays[t.day]&&t.teacherId;});
  var teacherIds={};relevantSlots.forEach(function(t){teacherIds[t.teacherId]=true;});

  if(!primaryTeacherId){var asg=state.courseAssignments.find(function(a){return a.semesterId===stu.currentSemesterId&&a.sectionId===stu.currentSectionId;});primaryTeacherId=asg?asg.teacherId:null;}
  var row={id:uid("lv"),studentId:session.linkedId,leaveType:data.leaveType,startDate:data.startDate,endDate:data.endDate,days:days,reason:data.reason,description:data.description||"",status:LEAVE_STATUS.WITH_TEACHER,currentOwnerRole:"teacher",currentOwnerId:primaryTeacherId,notifiedTeacherIds:Object.keys(teacherIds),teacherRemarks:"",chairmanRemarks:"",submittedAt:nowIso(),forwardedAt:null,decidedAt:null,timeline:[{at:nowIso(),action:"Submitted",by:session.userId,role:session.role,note:"Application submitted"}],programId:stu.programId,departmentId:stu.departmentId};
  updateColl("leaveApplications",function(a){return a.concat([row]);});
  Object.keys(teacherIds).forEach(function(tid){notifyUser(tid,"teacher","New Leave Request",stu.name+" submitted "+data.leaveType+" leave.",{relatedRecordId:row.id,relatedModule:"leaveApplications",relatedRoute:"#/app/leaves"});});

function findEligibleSubstitutes(timetableId,dateIso){
  var slot=state.timetable.find(function(t){return t.id===timetableId;});
  if(!slot)return [];
  var day=slot.day;
  return state.teachers.filter(function(t){
    if(t.id===slot.teacherId)return false;
    if(t.active===false)return false;
    if(t.programId&&slot.programId&&t.programId!==slot.programId)return false;
    var busy=state.timetable.some(function(other){
      if(other.id===slot.id||other.day!==day||other.teacherId!==t.id)return false;
      return t2m(slot.startTime)<t2m(other.endTime)&&t2m(slot.endTime)>t2m(other.startTime);
    });
    if(busy)return false;
    var onLeave=state.teacherLeaveApplications.some(function(l){return l.teacherId===t.id&&l.status===LEAVE_STATUS.APPROVED&&dateIso>=l.startDate&&dateIso<=l.endDate;});
    if(onLeave)return false;
    return true;
  });
}

function labelIconButtons(root){
  $$("button",root).forEach(function(b){
    if(!b.querySelector("svg")||b.textContent.trim())return;
    var label=b.getAttribute("aria-label")||b.getAttribute("title")||"";
    if(!label&&b.id==="menuBtn")label="Open navigation";
    if(!label&&b.hasAttribute("data-close"))label="Close dialog";
    if(!label&&b.hasAttribute("data-edit"))label="Edit";
    if(!label&&b.hasAttribute("data-del"))label="Delete";
    if(label){b.setAttribute("aria-label",label);b.title=label;if(b.hasAttribute("data-edit")||b.hasAttribute("data-del"))b.classList.add("icon-action");}
  });
}

  var url=URL.createObjectURL(blob);

}

}
function uploadFile(file){
  return new Promise(function(res,rej){
    if(file.size>5*1024*1024)return rej(new Error("Max 5 MB"));
    var r=new FileReader();
    r.onload=function(){res({fileName:file.name,fileType:file.type,fileSize:file.size,dataUrl:r.result});};
    r.onerror=rej;r.readAsDataURL(file);
  });
}

function renderRoute(){
  var path=location.hash||"#/login";
  var session=state.session;

  if(path!=="#/login"&&!session){
    try{localStorage.setItem(G.LAST_ROUTE_KEY, path);}catch(e){}
    if(location.hash!=="#/login") location.hash="#/login";
    return;
  }
  if((path==="#/login"||path==="#/"||path==="")&&session){
    var lastRoute=null;
    try{lastRoute=localStorage.getItem(G.LAST_ROUTE_KEY);}catch(e){}
    location.hash=(lastRoute && lastRoute.indexOf("#/app/")>=0) ? lastRoute : "#/app/dashboard";
    return;
  }
  if(session && path.indexOf("#/app/")>=0){
    try{localStorage.setItem(G.LAST_ROUTE_KEY, path);}catch(e){}
  }
  var match=findRoute(path);
  if(!match){
    if(!session){ if(location.hash!=="#/login") location.hash="#/login"; return; }
    mountPage(AppShell({session:session,activePath:path,title:"Not Found",childrenHtml:'<div class="empty-state"><h4>Page not found</h4><p>'+esc(path)+'</p></div>'}));
    return;
  }
  try{mountPage(match.route.page({session:session,params:match.params}));}
  catch(e){
    console.error("Route render error:",e);
    mountPage(AppShell({session:session,activePath:path,title:"Error",childrenHtml:'<div class="empty-state"><h4>Error loading page</h4><p>'+esc(e.message)+'</p></div>'}));
  }
}

  labelIconButtons(root);
}
